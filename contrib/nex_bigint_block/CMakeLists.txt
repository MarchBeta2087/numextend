# contrib/nex_bigint_block：分块 NTT 乘法（P1，issue #98）。
add_library(nex_bigint_block STATIC nex_bigint_block.c)
target_include_directories(nex_bigint_block PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nex_bigint_block PUBLIC nex)
if(MSVC)
    target_compile_options(nex_bigint_block PRIVATE /W4)
else()
    target_compile_options(nex_bigint_block PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(nex_block_test nex_block_test.c)
target_link_libraries(nex_block_test PRIVATE nex_bigint_block)
if(MSVC)
    target_compile_options(nex_block_test PRIVATE /W4)
else()
    target_compile_options(nex_block_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME block_ntt_selfcheck COMMAND nex_block_test)

add_executable(nex_block_bench nex_block_bench.c)
target_link_libraries(nex_block_bench PRIVATE nex_bigint_block)
if(MSVC)
    target_compile_options(nex_block_bench PRIVATE /W4)
else()
    target_compile_options(nex_block_bench PRIVATE -Wall -Wextra -Wpedantic)
endif()
