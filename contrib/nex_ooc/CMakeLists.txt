# contrib/nex_ooc：out-of-core 分块乘法原型（P1b-OOC，issue #104）。
# 由 -DNEX_ENABLE_BIGNUM=ON 一并构建；默认 OFF。
add_library(nex_ooc STATIC nex_ooc.c)
target_link_libraries(nex_ooc PUBLIC nex)
target_include_directories(nex_ooc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(nex_ooc PRIVATE /W4)
else()
    target_compile_options(nex_ooc PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(nex_ooc_test nex_ooc_test.c)
target_link_libraries(nex_ooc_test PRIVATE nex_ooc)
if(MSVC)
    target_compile_options(nex_ooc_test PRIVATE /W4)
else()
    target_compile_options(nex_ooc_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME nex_ooc_test COMMAND nex_ooc_test)

add_executable(nex_ooc_bench nex_ooc_bench.c)
target_link_libraries(nex_ooc_bench PRIVATE nex_ooc)
if(MSVC)
    target_compile_options(nex_ooc_bench PRIVATE /W4)
else()
    target_compile_options(nex_ooc_bench PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(nex_ooc_bench PRIVATE psapi)
endif()
